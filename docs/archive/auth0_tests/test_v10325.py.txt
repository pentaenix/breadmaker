def test_v10325_release_docs_review_plan_and_ui_assets_are_in_sync():
    version=(ROOT/'VERSION').read_text().strip()
    assert tuple(map(int,version.split('.'))) >= (10,3,25)
    assert f'Implemented through **{version}**' in (ROOT/'docs/BREADMAKER_REVIEW_IMPLEMENTATION_PLAN.md').read_text()
    index=(ROOT/'static/index.html').read_text()
    assert f'/static/css/profile/account-security.css?v={version}' in index
    assert f'/static/js/features/account/security.js?v={version}' in index
    assert 'Checkpoint 10.6' in (ROOT/'README.md').read_text()


def test_v10325_security_ui_is_account_auth0_only_and_local_runtime_stays_accountless():
    ui=(ROOT/'static/js/features/account/security.js').read_text()
    assert "runtimeSupports('account_sessions')" in ui
    assert "state.runtime?.auth?.auth_mode==='auth0'" in ui
    lowered=ui.lower()
    for forbidden in ('password_input','password_value','totp_secret','totp_code','recovery_code','access_token','refresh_token','client_secret'):
        assert forbidden not in lowered
    local=(ROOT/'src/runtime/local.py').read_text().lower()
    assert 'auth0' not in local and 'account_security' not in local
