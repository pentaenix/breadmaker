def test_v10335_account_link_state_wrong_session_does_not_burn_legitimate_transaction():
    from auth.accounts import ExternalIdentity
    from auth.linking import AccountLinkTransactionStore, AccountLinkingError
    from auth.oidc import OidcAuthorizationRequest

    now = 1_700_000_000.0
    store = AccountLinkTransactionStore(max_active=16, now=lambda: now)
    primary = ExternalIdentity('https://tenant.example.test/', 'auth0|primary', 'usr_primary', 'email_password', '', True, now, now)
    request = OidcAuthorizationRequest('https://tenant.example.test/authorize', 'link_state', 'nonce', 'challenge', '/', now + 300)
    store.bind(request, primary_user_id='usr_primary', primary_session_id='ses_primary', primary=primary)
    wrong = SimpleNamespace(user_id='usr_attacker', session_id='ses_attacker')
    right = SimpleNamespace(user_id='usr_primary', session_id='ses_primary')
    with pytest.raises(AccountLinkingError, match='ACCOUNT_LINK_PRIMARY_SESSION_CHANGED'):
        store.consume_for_principal('link_state', principal=wrong)
    assert store.consume_for_principal('link_state', principal=right).primary_user_id == 'usr_primary'


def test_v10335_step_up_state_wrong_session_does_not_burn_legitimate_transaction():
    from auth.oidc import OidcAuthorizationRequest
    from auth.security import AccountSecurityError, StepUpTransactionStore

    now = 1_700_000_000.0
    store = StepUpTransactionStore(max_transactions=16, max_grants=16, now=lambda: now)
    right = SimpleNamespace(user_id='usr_primary', session_id='ses_primary')
    wrong = SimpleNamespace(user_id='usr_attacker', session_id='ses_attacker')
    request = OidcAuthorizationRequest('https://tenant.example.test/authorize', 'step_state', 'nonce', 'challenge', '/', now + 300)
    store.bind_authorization(request, principal=right, purpose='sign_out_all_sessions')
    with pytest.raises(AccountSecurityError, match='SECURITY_STEP_UP_SESSION_MISMATCH'):
        store.consume_transaction_for_principal('step_state', principal=wrong)
    assert store.consume_transaction_for_principal('step_state', principal=right).user_id == 'usr_primary'

def test_v10335_link_and_step_up_stores_have_per_user_availability_ceilings():
    from auth.accounts import ExternalIdentity
    from auth.linking import AccountLinkTransactionStore, AccountLinkingError
    from auth.oidc import OidcAuthorizationRequest
    from auth.security import AccountSecurityError, StepUpTransactionStore

    now = 1_700_000_000.0
    primary = ExternalIdentity('https://tenant.example.test/', 'auth0|primary', 'usr_primary', 'email_password', '', True, now, now)
    links = AccountLinkTransactionStore(max_active=16, max_per_user=2, now=lambda: now)
    for index in range(2):
        req = OidcAuthorizationRequest('https://tenant.example/authorize', f'link_{index}', 'n', 'c', '/', now + 300)
        links.bind(req, primary_user_id='usr_primary', primary_session_id='ses_primary', primary=primary)
    with pytest.raises(AccountLinkingError, match='ACCOUNT_LINK_TRANSACTION_USER_CAPACITY_EXCEEDED'):
        req = OidcAuthorizationRequest('https://tenant.example/authorize', 'link_2', 'n', 'c', '/', now + 300)
        links.bind(req, primary_user_id='usr_primary', primary_session_id='ses_primary', primary=primary)

    step_up = StepUpTransactionStore(max_transactions=16, max_transactions_per_user=2, max_grants=16, now=lambda: now)
    principal = SimpleNamespace(user_id='usr_primary', session_id='ses_primary')
    for index in range(2):
        req = OidcAuthorizationRequest('https://tenant.example/authorize', f'step_{index}', 'n', 'c', '/', now + 300)
        step_up.bind_authorization(req, principal=principal, purpose='sign_out_all_sessions')
    with pytest.raises(AccountSecurityError, match='SECURITY_STEP_UP_TRANSACTION_USER_CAPACITY_EXCEEDED'):
        req = OidcAuthorizationRequest('https://tenant.example/authorize', 'step_2', 'n', 'c', '/', now + 300)
        step_up.bind_authorization(req, principal=principal, purpose='sign_out_all_sessions')

    grants = StepUpTransactionStore(
        max_transactions=16, max_transactions_per_user=16,
        max_grants=16, max_grants_per_user=2, now=lambda: now,
    )
    for index in range(2):
        tx = grants.bind_authorization(
            OidcAuthorizationRequest('https://tenant.example/authorize', f'grant_{index}', 'n', 'c', '/', now + 300),
            principal=principal, purpose='revoke_session', target=f'ses_target_{index}',
        )
        grants.issue_grant(tx, authenticated_at=now)
    tx = grants.bind_authorization(
        OidcAuthorizationRequest('https://tenant.example/authorize', 'grant_2', 'n', 'c', '/', now + 300),
        principal=principal, purpose='revoke_session', target='ses_target_2',
    )
    with pytest.raises(AccountSecurityError, match='SECURITY_STEP_UP_GRANT_USER_CAPACITY_EXCEEDED'):
        grants.issue_grant(tx, authenticated_at=now)

def test_v10335_hosted_rate_limiter_identity_store_is_hard_bounded():
    from security.hosted_guardrails import HostedRateLimiter

    now = [1_700_000_000.0]
    limiter = HostedRateLimiter(max_buckets=16, now=lambda: now[0])
    for index in range(16):
        assert limiter.allow(f'198.51.100.{index}', 'auth-start', limit=10) == (True, 0)
    allowed, retry = limiter.allow('203.0.113.1', 'auth-start', limit=10)
    assert allowed is False and retry >= 1
    now[0] += 61
    assert limiter.allow('203.0.113.1', 'auth-start', limit=10) == (True, 0)
