def test_v10333_oidc_state_is_single_use_and_return_paths_reject_browser_normalization_tricks():
    from auth.oidc import OidcError, OidcTransactionStore
    now=[1_700_000_000.0]
    store=OidcTransactionStore(ttl_seconds=60,now=lambda:now[0])
    tx=store.begin(return_path='/p/'+'a'*32+'?view=manuscript')
    assert store.consume(tx.state).return_path.startswith('/p/')
    with pytest.raises(OidcError,match='OIDC_STATE_INVALID_OR_REPLAYED'):
        store.consume(tx.state)
    for bad in ('//evil.example/x','https://evil.example/x','/\\evil.example/x','/%5Cevil.example/x','/%2F%2Fevil.example/x','/ok#fragment','/bad\x00x'):
        with pytest.raises(OidcError,match='OIDC_RETURN_PATH_INVALID'):
            store.begin(return_path=bad)

def test_v10333_session_replacement_invalidates_pre_auth_cookie_and_changes_identity():
    from auth.sessions import OpaqueSessionService
    sessions=OpaqueSessionService(auth_mode='auth0',environment='development')
    old,old_principal=sessions.issue('usr_attacker',auth_method='preauth')
    fresh,fresh_principal=sessions.replace(old,'usr_real',auth_method='google',authenticated_at=1.0)
    assert fresh != old and fresh_principal.session_id != old_principal.session_id
    with sessions.bind(old) as principal: assert principal is None
    with sessions.bind(fresh) as principal: assert principal and principal.user_id=='usr_real'
    assert old not in repr(sessions._records) and fresh not in repr(sessions._records)


def test_v10333_step_up_grant_is_bound_to_user_session_operation_target_and_single_use():
    from auth.oidc import OidcAuthorizationRequest
    from auth.security import AccountSecurityError, StepUpTransactionStore
    now=1_700_000_000.0
    store=StepUpTransactionStore(now=lambda:now)
    req=OidcAuthorizationRequest('https://auth.example/','state','nonce','challenge','/',now+300)
    principal=SimpleNamespace(user_id='usr_a',session_id='ses_a')
    tx=store.bind_authorization(req,principal=principal,purpose='unlink_identity',target='apple|x')
    store.issue_grant(tx,authenticated_at=now)
    for wrong in (
        SimpleNamespace(user_id='usr_b',session_id='ses_a'),
        SimpleNamespace(user_id='usr_a',session_id='ses_b'),
    ):
        with pytest.raises(AccountSecurityError): store.consume_grant(principal=wrong,purpose='unlink_identity',target='apple|x')
    with pytest.raises(AccountSecurityError): store.consume_grant(principal=principal,purpose='unlink_identity',target='apple|y')
    store.consume_grant(principal=principal,purpose='unlink_identity',target='apple|x')
    with pytest.raises(AccountSecurityError,match='SECURITY_STEP_UP_REQUIRED'):
        store.consume_grant(principal=principal,purpose='unlink_identity',target='apple|x')


def test_v10333_totp_replacement_does_not_delete_working_factor_if_new_enrollment_cannot_start():
    from auth.accounts import ExternalIdentity
    from auth.security import AccountSecurityError, HostedAuth0SecurityService, StepUpTransaction, StepUpTransactionStore
    from auth.auth0 import Auth0BoundaryConfig
    now=1_700_000_000.0
    primary=ExternalIdentity('https://tenant.example.test/','google-oauth2|p','usr_a','google','a@example.test',True,1,1)
    calls=[]
    class Accounts:
        def primary_identity_for_user(self,_): return primary
        def identities_for_user(self,_): return [primary]
    class Mgmt:
        def metadata(self): return {}
        def authentication_methods(self,_): return [{'id':'totp-1','type':'totp'}]
        def delete_authentication_method(self,*_): calls.append('delete')
    boundary=Auth0BoundaryConfig(environment='development',issuer='https://tenant.example.test/',client_id='client',application_origin='http://127.0.0.1:8766',client_secret_ref='env:S',callback_path='/auth/step-up/callback')
    store=StepUpTransactionStore(now=lambda:now)
    principal=SimpleNamespace(user_id='usr_a',session_id='ses_a')
    store.issue_grant(StepUpTransaction('s','usr_a','ses_a','replace_totp','totp-1','/',now,now+300),authenticated_at=now)
    service=HostedAuth0SecurityService(boundary=boundary,oidc=SimpleNamespace(),accounts=Accounts(),management=Mgmt(),intent_signer=SimpleNamespace(metadata=lambda:{}),step_up=store,now=lambda:now)
    def fail_start(**_): raise AccountSecurityError('SECURITY_PROVIDER_UNAVAILABLE')
    service.begin_step_up=fail_start
    with pytest.raises(AccountSecurityError,match='SECURITY_PROVIDER_UNAVAILABLE'):
        service.begin_totp_replacement_enrollment(principal=principal,method_id='totp-1')
    assert calls==[]
