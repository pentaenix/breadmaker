def test_v10334_initial_login_state_is_bound_to_the_browser_and_mismatch_does_not_burn_it():
    from auth.hosted_login import HostedAuth0LoginError, _LoginBrowserBindingStore
    now=[1_700_000_000.0]
    store=_LoginBrowserBindingStore(now=lambda:now[0],max_states=16)
    store.bind(state='state_abc',browser_binding='browser_A_abcdefghijklmnopqrstuvwxyz',expires_at=now[0]+300)
    with pytest.raises(HostedAuth0LoginError,match='AUTH0_LOGIN_BROWSER_BINDING_MISMATCH'):
        store.consume(state='state_abc',browser_binding='browser_B_abcdefghijklmnopqrstuvwxyz')
    store.consume(state='state_abc',browser_binding='browser_A_abcdefghijklmnopqrstuvwxyz')
    with pytest.raises(HostedAuth0LoginError,match='AUTH0_LOGIN_BROWSER_BINDING_INVALID'):
        store.consume(state='state_abc',browser_binding='browser_A_abcdefghijklmnopqrstuvwxyz')

def test_v10334_oidc_transaction_state_is_bounded_against_public_login_memory_abuse():
    from auth.oidc import OidcError, OidcTransactionStore
    store=OidcTransactionStore(ttl_seconds=60,max_active=16,now=lambda:1_700_000_000.0)
    for _ in range(16): store.begin(return_path='/')
    with pytest.raises(OidcError,match='OIDC_TRANSACTION_CAPACITY_EXCEEDED'):
        store.begin(return_path='/')
    with pytest.raises(OidcError,match='OIDC_STATE_INVALID_OR_REPLAYED'):
        store.consume('x'*257)

def test_v10334_link_and_step_up_state_stores_have_hard_capacities():
    from types import SimpleNamespace
    from auth.accounts import ExternalIdentity
    from auth.linking import AccountLinkTransactionStore, AccountLinkingError
    from auth.oidc import OidcAuthorizationRequest
    from auth.security import AccountSecurityError, StepUpTransactionStore

    now=1_700_000_000.0
    primary=ExternalIdentity('https://issuer/','auth0|primary','usr_primary','email_password','p@example.test',True,now,now)
    link=AccountLinkTransactionStore(max_active=16,max_per_user=16,now=lambda:now)
    for index in range(16):
        state=f'link_state_{index}'
        req=OidcAuthorizationRequest('https://issuer/authorize',state,'nonce','challenge','/',now+300)
        link.bind(req,primary_user_id='usr_primary',primary_session_id='ses_primary',primary=primary)
    with pytest.raises(AccountLinkingError,match='ACCOUNT_LINK_TRANSACTION_CAPACITY_EXCEEDED'):
        req=OidcAuthorizationRequest('https://issuer/authorize','link_state_overflow','nonce','challenge','/',now+300)
        link.bind(req,primary_user_id='usr_primary',primary_session_id='ses_primary',primary=primary)

    principal=SimpleNamespace(user_id='usr_primary',session_id='ses_primary')
    step=StepUpTransactionStore(max_transactions=16,max_transactions_per_user=16,max_grants=16,now=lambda:now)
    transactions=[]
    for index in range(16):
        req=OidcAuthorizationRequest('https://issuer/authorize',f'step_state_{index}','nonce','challenge','/',now+300)
        transactions.append(step.bind_authorization(req,principal=principal,purpose='unlink_identity',target=f'auth0|secondary_{index}'))
    with pytest.raises(AccountSecurityError,match='SECURITY_STEP_UP_TRANSACTION_CAPACITY_EXCEEDED'):
        req=OidcAuthorizationRequest('https://issuer/authorize','step_state_overflow','nonce','challenge','/',now+300)
        step.bind_authorization(req,principal=principal,purpose='unlink_identity',target='auth0|overflow')
    for tx in transactions:
        step.issue_grant(tx,authenticated_at=now)
    with pytest.raises(AccountSecurityError,match='SECURITY_STEP_UP_GRANT_CAPACITY_EXCEEDED'):
        overflow=transactions[0].__class__('step_state_extra','usr_primary','ses_primary','unlink_identity','auth0|extra','/',now,now+300)
        step.issue_grant(overflow,authenticated_at=now)

def test_v10334_opaque_session_registry_is_bounded_without_evicting_other_users():
    from auth.sessions import AuthenticationError, OpaqueSessionService
    sessions=OpaqueSessionService(auth_mode='auth0',environment='development',max_sessions_per_user=8,max_sessions_total=256)
    first=[]
    for index in range(9):
        token,_=sessions.issue('usr_a',auth_method=f'm{index}')
        first.append(token)
    # Per-user overflow retires only that user's oldest session.
    with sessions.bind(first[0]) as principal: assert principal is None
    with sessions.bind(first[-1]) as principal: assert principal and principal.user_id=='usr_a'
    other,_=sessions.issue('usr_b',auth_method='google')
    for index in range(16): sessions.issue('usr_a',auth_method=f'x{index}')
    with sessions.bind(other) as principal: assert principal and principal.user_id=='usr_b'
    assert sessions.metadata()['bounded_session_registry'] is True
