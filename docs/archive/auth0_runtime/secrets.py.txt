from __future__ import annotations

"""Hosted secret-resolution boundary.

The normal local runtime never instantiates a hosted secret loader. Development Auth0 may
read explicit ``env:NAME`` references. Staging resolves only explicit Google Secret Manager
version resource names using the workload's Application Default Credentials.
"""

import os
import re
from dataclasses import dataclass
from typing import Callable, Iterable


class HostedSecretError(RuntimeError):
    """Stable non-secret failure from the hosted secret channel."""


_ENV_NAME = re.compile(r"^[A-Z][A-Z0-9_]{0,127}$")
_SECRET_RESOURCE = re.compile(
    r"^projects/(?P<project>[a-z][a-z0-9-]{4,28}[a-z0-9])/secrets/"
    r"(?P<secret>[A-Za-z0-9_-]{1,255})/versions/(?P<version>latest|[1-9][0-9]{0,18})$"
)
_STAGING_SECRET_IDS = frozenset({
    'breadmaker-auth0-web-client-secret',
    'breadmaker-auth0-link-client-secret',
    'breadmaker-auth0-security-client-secret',
    'breadmaker-auth0-security-intent-secret',
})


def _clean_reference(reference: str) -> str:
    value = str(reference or "").strip()
    if not value or len(value) > 512 or any(ch.isspace() for ch in value):
        raise HostedSecretError("HOSTED_SECRET_REFERENCE_INVALID")
    return value


def development_environment_secret_loader(reference: str) -> str:
    """Resolve ``env:NAME`` only in the explicit localhost Auth0 sandbox."""
    ref = _clean_reference(reference)
    prefix, separator, name = ref.partition(":")
    if separator != ":" or prefix != "env" or _ENV_NAME.fullmatch(name) is None:
        raise HostedSecretError("AUTH0_DEV_SECRET_REFERENCE_INVALID")
    value = str(os.environ.get(name) or "")
    if not value:
        raise HostedSecretError("AUTH0_DEV_SECRET_UNAVAILABLE")
    return value


@dataclass
class GoogleSecretManagerLoader:
    """Resolve one exact Secret Manager version through ADC.

    The reference is configuration, never a secret value. Restricting references to the
    configured staging project prevents a compromised environment variable from turning
    Breadmaker into a cross-project secret reader if the service account later gains
    unrelated access by mistake.
    """

    project_id: str
    client_factory: Callable[[], object] | None = None
    allowed_secret_ids: Iterable[str] | None = None

    def __post_init__(self) -> None:
        project = str(self.project_id or "").strip()
        if not project or len(project) > 128:
            raise HostedSecretError("SECRET_MANAGER_PROJECT_INVALID")
        self.project_id = project
        if self.allowed_secret_ids is None:
            self._allowed_secret_ids = None
        else:
            values = frozenset(str(value or '').strip() for value in self.allowed_secret_ids)
            if not values or any(not value or len(value) > 255 for value in values):
                raise HostedSecretError("SECRET_MANAGER_ALLOWLIST_INVALID")
            self._allowed_secret_ids = values

    def _client(self):
        if self.client_factory is not None:
            return self.client_factory()
        try:
            from google.cloud import secretmanager
        except Exception as exc:
            raise HostedSecretError("SECRET_MANAGER_CLIENT_UNAVAILABLE") from exc
        try:
            return secretmanager.SecretManagerServiceClient()
        except Exception as exc:
            raise HostedSecretError("SECRET_MANAGER_CLIENT_UNAVAILABLE") from exc

    def __call__(self, reference: str) -> str:
        ref = _clean_reference(reference)
        prefix, separator, resource = ref.partition(":")
        if separator != ":" or prefix != "secretmanager":
            raise HostedSecretError("SECRET_MANAGER_REFERENCE_INVALID")
        match = _SECRET_RESOURCE.fullmatch(resource)
        if match is None:
            raise HostedSecretError("SECRET_MANAGER_REFERENCE_INVALID")
        if match.group("project") != self.project_id:
            raise HostedSecretError("SECRET_MANAGER_PROJECT_MISMATCH")
        if self._allowed_secret_ids is not None and match.group("secret") not in self._allowed_secret_ids:
            raise HostedSecretError("SECRET_MANAGER_SECRET_NOT_ALLOWED")
        try:
            response = self._client().access_secret_version(request={"name": resource})
            payload = bytes(response.payload.data)
            value = payload.decode("utf-8")
        except HostedSecretError:
            raise
        except Exception as exc:
            # Provider exception text can contain resource and transport details. Keep
            # it chained for server diagnostics, but publish only the stable code.
            raise HostedSecretError("SECRET_MANAGER_SECRET_UNAVAILABLE") from exc
        if not value or len(value) > 65536 or "\x00" in value:
            raise HostedSecretError("SECRET_MANAGER_SECRET_INVALID")
        return value


def build_hosted_secret_loader(environment: str):
    env = str(environment or "").strip().lower()
    if env == "development":
        return development_environment_secret_loader
    if env == "staging":
        project = str(os.environ.get("BREADMAKER_GCP_PROJECT_ID") or "").strip()
        if not project:
            raise HostedSecretError("SECRET_MANAGER_PROJECT_REQUIRED")
        return GoogleSecretManagerLoader(project, allowed_secret_ids=_STAGING_SECRET_IDS)
    if env == "production":
        raise HostedSecretError("HOSTED_PRODUCTION_NOT_ENABLED")
    raise HostedSecretError("HOSTED_SECRET_ENVIRONMENT_INVALID")


__all__ = [
    "HostedSecretError",
    "GoogleSecretManagerLoader",
    "development_environment_secret_loader",
    "_STAGING_SECRET_IDS",
    "build_hosted_secret_loader",
]
